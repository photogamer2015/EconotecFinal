<script>
document.getElementById('contacto-econotec').addEventListener('submit', async e => {
  e.preventDefault();
  const datos = Object.fromEntries(new FormData(e.target));
  const r = await fetch('{{ base }}{% url "crm:api_web" %}', {
    method: 'POST',
    headers: {'Content-Type': 'application/json', 'X-CRM-Token': 'TU_TOKEN'},
    body: JSON.stringify(datos)
  });
  e.target.innerHTML = r.ok ? '<p>¡Gracias! Una asesora de Econotec le contactará pronto.</p>' : '<p>No se pudo enviar, escríbanos por WhatsApp.</p>';
});
</script>
